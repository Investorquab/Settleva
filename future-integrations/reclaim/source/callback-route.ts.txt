import { NextResponse } from "next/server";
import { ReclaimSessionStore } from "../../../../lib/reclaim-session-store.js";
import { JsonRequestBodyError, readJsonRequestBody } from "../../../../lib/request-body.js";
import { ReclaimVerificationError, verifyReclaimAndAttest } from "../../../../lib/reclaim-verification.js";
import { resolveCallbackFailure, resolveVerificationCommit } from "../../../../lib/reclaim-decision.js";

export const runtime = "nodejs";

const MAX_CALLBACK_BODY_BYTES = 1024 * 1024;
const TERMINAL_CONDITION_FAILURE = "Verified evidence did not satisfy the committed payment condition.";

function privateJson(body: unknown, init?: ResponseInit) {
  const headers = new Headers(init?.headers);
  headers.set("Cache-Control", "no-store");
  headers.set("X-Content-Type-Options", "nosniff");
  return NextResponse.json(body, {...init, headers});
}

function parseSessionId(proof: unknown): string | null {
  if (!proof || typeof proof !== "object" || Array.isArray(proof)) return null;
  const claimData = (proof as Record<string, unknown>).claimData;
  if (!claimData || typeof claimData !== "object" || Array.isArray(claimData)) return null;
  const context = (claimData as Record<string, unknown>).context;
  if (typeof context !== "string" || context.length > 16_384) return null;
  try {
    const parsed = JSON.parse(context) as Record<string, unknown> | null;
    return parsed && typeof parsed === "object" && typeof parsed.reclaimSessionId === "string"
      && parsed.reclaimSessionId.length > 0 && parsed.reclaimSessionId.length <= 256
      ? parsed.reclaimSessionId
      : null;
  } catch {
    return null;
  }
}

function publicVerificationError(error: unknown, retryable: boolean, terminal: boolean): string {
  if (terminal) return TERMINAL_CONDITION_FAILURE;
  if (error instanceof ReclaimVerificationError && error.code === "REPLAY") {
    return "This proof or session has already been accepted.";
  }
  if (retryable) return "Proof verification is temporarily unavailable. The session remains retryable.";
  return "The submitted proof could not be verified. The session remains pending; retry with a valid fresh proof.";
}

export async function POST(request: Request) {
  const databaseUrl = process.env.DATABASE_URL;
  if (!databaseUrl) return privateJson({error:"Replay database is not configured."},{status:503});

  let sessions: ReclaimSessionStore | undefined;
  try {
    const raw = await readJsonRequestBody<unknown>(request, MAX_CALLBACK_BODY_BYTES);
    const proofs = Array.isArray(raw) ? raw : [raw];
    if (proofs.length !== 1) return privateJson({error:"Exactly one Reclaim proof is required."},{status:400});

    const sessionId = parseSessionId(proofs[0]);
    if (!sessionId) return privateJson({error:"Reclaim session binding is missing or malformed."},{status:400});

    sessions = new ReclaimSessionStore(databaseUrl);
    const session = await sessions.get(sessionId);
    if (!session) return privateJson({error:"Unknown Reclaim session."},{status:404});
    if (session.status === "verified") return privateJson({received:true,verified:true,sessionId});
    if (session.status === "failed") {
      return privateJson({received:true,verified:false,sessionId,error:TERMINAL_CONDITION_FAILURE},{status:409});
    }

    try {
      const result = await verifyReclaimAndAttest({
        proofs,
        sessionId,
        condition:session.condition,
        expectedPaymentId:session.paymentId,
        expectedConditionHash:session.conditionHash as `0x${string}`,
        expectedProviderId:session.providerId,
        expectedProviderVersion:session.providerVersion
      });

      let transitioned: boolean;
      try {
        transitioned = await sessions.markVerified(sessionId,result.proof,result.proofIdentifier,result.verificationSignature);
      } catch {
        const currentStatus = (await sessions.get(sessionId))?.status ?? null;
        if (currentStatus === "verified") return privateJson({received:true,verified:true,sessionId});
        throw new ReclaimVerificationError("DATABASE", "Could not persist the verified session state.");
      }

      const commitStatus = resolveVerificationCommit(
        transitioned,
        transitioned ? "verified" : (await sessions.get(sessionId))?.status ?? null
      );
      if (commitStatus === "already-committed") return privateJson({received:true,verified:true,sessionId});
      if (commitStatus === "conflict") {
        return privateJson({received:true,verified:false,sessionId,error:"Reclaim session changed state before verification could be committed."},{status:409});
      }
      return privateJson({received:true,verified:true,sessionId});
    } catch (error) {
      const code = error instanceof ReclaimVerificationError ? error.code : null;
      const terminal = code === "CONDITION_FAILED";
      const retryable = code === "DATABASE" || code === "REPLAY" || code === "CHAIN" || code === "PROVIDER";
      const currentStatus = (await sessions.get(sessionId))?.status ?? null;
      const failureStatus = resolveCallbackFailure(terminal, currentStatus);

      if (failureStatus === "already-verified") return privateJson({received:true,verified:true,sessionId});

      if (failureStatus === "mark-failed") {
        let marked: boolean;
        try {
          marked = await sessions.markFailed(sessionId, TERMINAL_CONDITION_FAILURE);
        } catch {
          return privateJson({received:true,verified:false,error:"Could not persist verification state. Retry the callback."},{status:503});
        }
        if (!marked) {
          const latestStatus = (await sessions.get(sessionId))?.status ?? null;
          if (latestStatus === "verified") return privateJson({received:true,verified:true,sessionId});
          return privateJson({received:true,verified:false,error:"Reclaim session changed state during verification."},{status:409});
        }
      }

      const status = terminal ? 422
        : code === "REPLAY" ? 409
        : retryable ? 503
        : error instanceof ReclaimVerificationError && error.code === "INVALID" ? 400
        : 503;
      return privateJson({received:true,verified:false,sessionId,error:publicVerificationError(error,retryable,terminal)},{status});
    }
  } catch (error) {
    if (error instanceof JsonRequestBodyError) return privateJson({error:error.message},{status:error.status});
    return privateJson({error:"Could not process the Reclaim callback. Retry later."},{status:503});
  } finally {
    if (sessions) await sessions.close().catch(() => undefined);
  }
}
