import type { PaymentCondition, VerifiedClaim } from "@settleva/conditions";
import { extractGitHubDeploymentClaims, GITHUB_DEPLOYMENT_CLAIM_FIELDS, isGitHubDeploymentCondition } from "@settleva/providers";

export interface ReclaimEvidenceAdapter {
  readonly providerId: string;
  readonly providerVersion: string;
  buildProviderParams(condition: PaymentCondition): Record<string, string>;
  normalizeClaims(extractedParameters: Record<string, unknown>, condition: PaymentCondition): VerifiedClaim[];
}
interface MappedAdapterConfiguration {
  readonly providerId: string;
  readonly providerVersion: string;
  readonly kind: "mapped";
  /** Canonical condition field -> Reclaim dynamic request parameter name. */
  readonly requestParameters: Readonly<Record<string, string>>;
  /** Reclaim extracted-parameter name -> canonical condition field. */
  readonly claimFields: Readonly<Record<string, string>>;
}
export class ReclaimAdapterConfigurationError extends Error {
  constructor(message: string) { super(message); this.name = "ReclaimAdapterConfigurationError"; }
}
function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}
function requiredText(value: unknown, label: string): string {
  if (typeof value !== "string" || !value.trim() || value !== value.trim()) throw new ReclaimAdapterConfigurationError(label + " must be a non-empty trimmed string.");
  return value;
}
function parseFieldMap(value: unknown, label: string, allowEmpty: boolean): Record<string, string> {
  if (!isRecord(value)) {
    if (allowEmpty && value === undefined) return {};
    throw new ReclaimAdapterConfigurationError(label + " must be an object of field mappings.");
  }
  const result: Record<string, string> = {};
  const values = new Set<string>();
  for (const [key, rawValue] of Object.entries(value)) {
    const source = requiredText(key, label + " source field");
    const target = requiredText(rawValue, label + " target field");
    if (["__proto__", "constructor", "prototype"].includes(source) || ["__proto__", "constructor", "prototype"].includes(target)) throw new ReclaimAdapterConfigurationError(label + " contains a reserved field name.");
    if (Object.hasOwn(result, source)) throw new ReclaimAdapterConfigurationError(label + " contains a duplicate source field.");
    if (values.has(target)) throw new ReclaimAdapterConfigurationError(label + " maps multiple source fields to the same target.");
    result[source] = target; values.add(target);
  }
  if (!allowEmpty && Object.keys(result).length === 0) throw new ReclaimAdapterConfigurationError(label + " cannot be empty.");
  return result;
}
function parseRegistry(env: Record<string, string | undefined>): MappedAdapterConfiguration[] {
  const raw = env.SETTLEVA_RECLAIM_ADAPTERS?.trim();
  if (!raw) return [];
  let parsed: unknown;
  try { parsed = JSON.parse(raw) as unknown; } catch { throw new ReclaimAdapterConfigurationError("SETTLEVA_RECLAIM_ADAPTERS must contain valid JSON."); }
  if (!Array.isArray(parsed)) throw new ReclaimAdapterConfigurationError("SETTLEVA_RECLAIM_ADAPTERS must be a JSON array.");
  const identities = new Set<string>();
  return parsed.map((item, index) => {
    if (!isRecord(item) || item.kind !== "mapped") throw new ReclaimAdapterConfigurationError("Adapter " + index + " must use the supported kind \"mapped\".");
    const providerId = requiredText(item.providerId, "Adapter " + index + " providerId");
    const providerVersion = requiredText(item.providerVersion, "Adapter " + index + " providerVersion");
    const identity = providerId + "\u0000" + providerVersion;
    if (identities.has(identity)) throw new ReclaimAdapterConfigurationError("SETTLEVA_RECLAIM_ADAPTERS contains a duplicate provider ID/version.");
    identities.add(identity);
    const requestParameters = parseFieldMap(item.requestParameters, "Adapter " + index + " requestParameters", true);
    const claimFields = parseFieldMap(item.claimFields, "Adapter " + index + " claimFields", false);
    const requestNames = new Set(Object.values(requestParameters));
    if (Object.keys(claimFields).some((field) => requestNames.has(field))) throw new ReclaimAdapterConfigurationError("Adapter " + index + " cannot treat a dynamic request parameter as verified evidence; use an independently extracted response field.");
    return {providerId, providerVersion, kind:"mapped", requestParameters, claimFields};
  });
}
function makeMappedAdapter(config: MappedAdapterConfiguration): ReclaimEvidenceAdapter {
  return {
    providerId:config.providerId, providerVersion:config.providerVersion,
    buildProviderParams(condition) {
      const params: Record<string,string> = {};
      for (const [conditionField, parameterName] of Object.entries(config.requestParameters)) {
        const matches = condition.claims.filter((claim) => claim.field === conditionField);
        if (matches.length !== 1 || matches[0].operator !== "equals" || typeof matches[0].value !== "string") throw new ReclaimAdapterConfigurationError("The committed condition must contain exactly one equals claim for request field " + conditionField + ".");
        if (!/^[A-Za-z_][A-Za-z0-9_.-]{0,127}$/.test(parameterName) || Object.hasOwn(params,parameterName)) throw new ReclaimAdapterConfigurationError("The adapter contains an invalid or duplicate Reclaim request parameter name.");
        params[parameterName] = matches[0].value;
      }
      return params;
    },
    normalizeClaims(extractedParameters, condition) {
      const expectedFields = new Set(condition.claims.map((claim) => claim.field));
      const seenFields = new Set<string>();
      const claims: VerifiedClaim[] = [];
      for (const [sourceField, rawValue] of Object.entries(extractedParameters)) {
        const canonicalField = config.claimFields[sourceField];
        if (!canonicalField || !expectedFields.has(canonicalField)) continue;
        if (typeof rawValue !== "string" && typeof rawValue !== "number" && typeof rawValue !== "boolean") throw new ReclaimAdapterConfigurationError("A mapped evidence field is not a scalar value.");
        if (seenFields.has(canonicalField)) throw new ReclaimAdapterConfigurationError("The provider returned duplicate values for one canonical evidence field.");
        seenFields.add(canonicalField); claims.push({field:canonicalField,value:String(rawValue)});
      }
      return claims;
    }
  };
}
const GITHUB_RAW_FIELD_MAP: Readonly<Record<string,string>> = {
  github_repo_full_name:GITHUB_DEPLOYMENT_CLAIM_FIELDS.repository,
  github_deployment_ref:GITHUB_DEPLOYMENT_CLAIM_FIELDS.ref,
  github_deployment_sha:GITHUB_DEPLOYMENT_CLAIM_FIELDS.sha,
  github_deployment_environment:GITHUB_DEPLOYMENT_CLAIM_FIELDS.environment,
  github_deployment_status:GITHUB_DEPLOYMENT_CLAIM_FIELDS.status
};
function makeGitHubAdapter(providerId: string, providerVersion: string): ReclaimEvidenceAdapter {
  return {
    providerId, providerVersion,
    buildProviderParams(condition) {
      const deployment = extractGitHubDeploymentClaims(condition.claims);
      if (!deployment) throw new ReclaimAdapterConfigurationError("GitHub deployment conditions must include repository, ref, commit SHA, environment and status.");
      return {
        github_repo_full_name:deployment.repository, github_deployment_ref:deployment.ref,
        github_deployment_sha:deployment.sha, github_deployment_environment:deployment.environment,
        github_deployment_status:deployment.status
      };
    },
    normalizeClaims(extractedParameters, condition) {
      const expectedFields = new Set(condition.claims.map((claim) => claim.field));
      const seenFields = new Set<string>(), claims: VerifiedClaim[] = [];
      const githubFields = new Set<string>(Object.values(GITHUB_DEPLOYMENT_CLAIM_FIELDS));
      for (const [sourceField, rawValue] of Object.entries(extractedParameters)) {
        const field = GITHUB_RAW_FIELD_MAP[sourceField] ?? (githubFields.has(sourceField) ? sourceField : undefined);
        if (!field || !expectedFields.has(field)) continue;
        if (typeof rawValue !== "string" && typeof rawValue !== "number" && typeof rawValue !== "boolean") throw new ReclaimAdapterConfigurationError("A mapped GitHub evidence field is not a scalar value.");
        if (seenFields.has(field)) throw new ReclaimAdapterConfigurationError("GitHub evidence contains duplicate canonical fields.");
        seenFields.add(field); claims.push({field,value:String(rawValue)});
      }
      if (isGitHubDeploymentCondition(condition) && !extractGitHubDeploymentClaims(claims)) throw new ReclaimAdapterConfigurationError("GitHub deployment evidence is incomplete or malformed.");
      return claims;
    }
  };
}
/** Resolve solely by the exact provider ID/version committed in the policy. */
export function resolveReclaimEvidenceAdapter(condition: PaymentCondition, env: Record<string,string|undefined> = process.env): ReclaimEvidenceAdapter {
  const configured = parseRegistry(env);
  const match = configured.find((entry) => entry.providerId === condition.provider && entry.providerVersion === condition.providerVersion);
  if (match) return makeMappedAdapter(match);
  if (isGitHubDeploymentCondition(condition) && env.RECLAIM_PROVIDER_ID === condition.provider && env.RECLAIM_PROVIDER_VERSION === condition.providerVersion) return makeGitHubAdapter(condition.provider,condition.providerVersion);
  throw new ReclaimAdapterConfigurationError("No evidence adapter is registered for the exact provider ID/version in this condition.");
}
export function hasConfiguredReclaimAdapter(env: Record<string,string|undefined> = process.env): boolean {
  return parseRegistry(env).length > 0 || Boolean(env.RECLAIM_PROVIDER_ID?.trim() && env.RECLAIM_PROVIDER_VERSION?.trim());
}
